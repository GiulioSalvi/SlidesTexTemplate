% Native frame titles grow vertically when a title wraps to a second line.
\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \begin{beamercolorbox}[
    wd=\paperwidth,leftskip=0.05\paperwidth,rightskip=0.05\paperwidth
  ]{frametitle}
    \vskip0.10cm
    \usebeamerfont{frametitle}\strut\insertframetitle\strut\par
    \ifx\insertframesubtitle\empty\else
      {\usebeamerfont{framesubtitle}\usebeamercolor[fg]{framesubtitle}%
       \insertframesubtitle\par}%
    \fi
    \vskip0.12cm
  \end{beamercolorbox}%
}
\setbeamertemplate{footline}{%
  \leavevmode\hbox{%
    \begin{beamercolorbox}[
      wd=0.82\paperwidth,ht=0.40cm,dp=0.16cm,leftskip=0.05\paperwidth
    ]{footline left}%
      \usebeamerfont{footline}\strut\begin{NoHyper}\SlidesFooter\end{NoHyper}
    \end{beamercolorbox}%
    \begin{beamercolorbox}[
      wd=0.18\paperwidth,ht=0.40cm,dp=0.16cm,center
    ]{footline right}%
      \usebeamerfont{footline}\strut\insertframenumber\,/\,\inserttotalframenumber
    \end{beamercolorbox}%
  }%
}

% A plain cover with independent top and bottom identity areas. Title,
% subtitle and rule flow together, preserving spacing when the title wraps.
\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north west,inner sep=0,text=SlidesAccent,font=\usebeamerfont{deck label}]
      at ([xshift=0.05\paperwidth,yshift=-0.04\paperheight]current page.north west)
      {\SlidesDeck};
    \node[anchor=north east,inner sep=0]
      at ([xshift=-0.05\paperwidth,yshift=-0.03\paperheight]current page.north east)
      {\SlidesCoverLogo};
    \node[anchor=north west,inner sep=0,text width=0.90\paperwidth,align=left]
      at ([xshift=0.05\paperwidth,yshift=-0.17\paperheight]current page.north west)
      {%
        {\usebeamerfont{title}\usebeamercolor[fg]{title}\inserttitle\par}%
        \vskip0.18cm
        {\usebeamerfont{subtitle}\usebeamercolor[fg]{subtitle}\insertsubtitle\par}%
        \vskip0.55cm
        {\color{SlidesAccent}\rule{\linewidth}{1pt}}%
      };
    \node[anchor=south west,inner sep=0,text width=0.90\paperwidth,align=left]
      at ([xshift=0.05\paperwidth,yshift=0.16\paperheight]current page.south west)
      {%
        {\usebeamerfont{author}\usebeamercolor[fg]{author}\insertauthor\par}%
        \ifdefempty{\SlidesTeam}{}{%
          {\usebeamerfont{institute}\color{SlidesMuted}\SlidesTeam\par}}%
        {\usebeamerfont{institute}\usebeamercolor[fg]{institute}\insertinstitute\par}%
        {\footnotesize\color{SlidesMuted}\insertdate\par}%
      };
  \end{tikzpicture}%
}

% Apply the deck default before each overlay's body. Native Beamer transition
% commands in that body can override it. Only the first overlay transitions
% by default; progressive reveals remain immediate PDF page changes.
\NewDocumentCommand{\SlideTransition}{D<>{1} O{} m}{%
  \edef\SlidesTransitionName{#3}%
  \ifdefstring{\SlidesTransitionName}{none}{\transreplace<#1>}{%
    \ifcsname trans\SlidesTransitionName\endcsname
      \csname trans\SlidesTransitionName\endcsname<#1>[%
        duration=\SlidesTransitionDuration,#2]%
    \else
      \PackageError{slides}{Unknown transition: #3}%
        {Use none, fade, dissolve, wipe, push, cover or another Beamer transition.}%
    \fi
  }%
}
\AddToHook{env/beamer@framepauses/begin}{%
  \SlideTransition{\SlidesDefaultTransition}%
}
